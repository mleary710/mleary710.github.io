Data Source

The data available has been cleaned for the most part, but looking at the last code output we see that the data is broken down into ‘rounds’. I aggregated the data to be broken down into fights using multiple variables to group observations and also made the following modifications.

Cleaning

# Remove 'attendance' variable
ufc_cleaner <- ufc_stats %>%
  select(-attendance)

Looking at the first few rows of ufc_stats we can see that all the values for ‘attendance’ are null. This isn’t a variable I’m concerned about so I removed it from the data set entirely.

# Create fight-end indicator
ufc_cleaner <- ufc_cleaner %>%
  mutate(fight_end = paste0("Round ", last_round, " - ", time))

This variable will still show how many rounds each fight lasts followed by the time left in the round.

# Create a unique opponent lookup table
opponent_lookup <- ufc_cleaner %>%
  select(id, fighter) %>%
  # Ensure unique fighters with fight 'id'
  distinct() %>%            
  group_by(id) %>%
  # Only keep fights with 2 fighters
  filter(n() == 2) %>%     
  # Reverse fighters for correct pairing
  mutate(opponent = rev(fighter)) %>% 
  ungroup()

This table helps add an ‘opponent’ variable to each observation that I then joined with the clean ufc_cleaner table

# Left join to add 'opponent_lookup' to 'ufc_cleaner'
ufc_cleaned <- ufc_cleaner %>%
  left_join(opponent_lookup, by = c("id", "fighter"))

Aggreate

ufc_agg <- ufc_cleaned %>%
  group_by(
    id, fighter, opponent, winner, result, event, fight_date, location, fight_end, scheduled_rounds
  ) %>%
  summarize(
    weight_classes = paste(unique(weight_class), collapse = ", "),
    knockdowns = sum(knockdowns, na.rm = TRUE),
    sig_strikes_landed = sum(significant_strikes_landed, na.rm = TRUE),
    sig_strikes_attempted = sum(significant_strikes_attempted, na.rm = TRUE),
    takedown_landed = sum(takedown_successful, na.rm = TRUE),
    takedown_attempted = sum(takedown_attempted, na.rm = TRUE),
    submission_attempt = sum(submission_attempt, na.rm = TRUE),
    head_landed = sum(head_landed, na.rm = TRUE),
    head_attempted = sum(head_attempted, na.rm = TRUE),
    body_landed = sum(body_landed, na.rm = TRUE),
    body_attempted = sum(body_attempted, na.rm = TRUE),
    leg_landed = sum(leg_landed, na.rm = TRUE),
    leg_attempted = sum(leg_attempted, na.rm = TRUE),
    clinch_landed = sum(clinch_landed, na.rm = TRUE),
    clinch_attempted = sum(clinch_attempted, na.rm = TRUE),

Average Number of Signigicant Strikes

I wanted to see if Max Holloway’s large total_significant_strike was attributed to him having a significantly larger number of fights while also looking at the average number of significant strikes a fight with other fighters.

# Summarize total significant strikes and number of fights to create `avg_n_sstrikes`
most_sstrikes <- ufc_agg %>%
  group_by(fighter) %>%
  summarize(
    total_fights = n(),
    total_significant_strikes = sum(sig_strikes_landed)
  ) %>%
  mutate(avg_n_sstrikes = round(total_significant_strikes / total_fights, 2)) %>%
  arrange(desc(total_significant_strikes))
